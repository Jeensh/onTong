public EdgingSpecEntity findSpec(String cmpCd, String orgCd, String edgingGroupCd) {
    // 1) 정확매칭
    Optional<EdgingSpecJpo> exact = specRepository.findById(
        new EdgingSpecPK(cmpCd, orgCd, edgingGroupCd));
